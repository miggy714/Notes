Cm   <- 2.27
m    <- 0.150
f    <- 0.065
Ct   <- 1500
nt   <- 5000
Lf   <- 0.9
Cc   <- 180000
two_hr <- 5 * 365 * 24
Coh  <- 100
n_dot <- 0.5

cost_per_part <- function(n) {
  Cm * m / (1 - f) +
    (Ct / n) * ceiling(n / nt) +
    (Lf / n_dot) * (Cc / two_hr) +
    (Coh / n_dot)
}
